# --description--

يجب أن يتمكن المستخدمون من التحقق من صندوق الوارد الخاص بهم، وقراءة الرسائل الإلكترونية، وحذف الرسائل الإلكترونية مباشرةً من خلال كائن `User` الخاص بهم.

للتأكد من صندوق الوارد، أضف طريقة تسمى `check_inbox` إلى صنف `User`. يجب أن تطبع هذه الطريقة قسمًا علويًا مخصصًا باسم المستخدم ثم تعرض جميع رسائله الإلكترونية.

يجب تنسيق القسم العلوي كالتالي: `\nName's Inbox:`، حيث يُستبدَل `Name` باسم المستخدم.

بعد طباعة القسم العلوي، استدعِ طريقة `list_emails()` على صندوق الوارد الخاص بالمستخدم.

class Inbox:
    def __init__(self):
        self.emails = []

    def receive_email(self, email):
        self.emails.append(email)

    def list_emails(self):
        if not self.emails:
            print('Your inbox is empty.\n')
            return
        print('\nYour Emails:')
        for i, email in enumerate(self.emails, start=1):
            print(f'{i:>2}. {email}')

    def read_email(self, index):
        if not self.emails:
            print('Inbox is empty.\n')
            return
        actual_index = index - 1
        if actual_index < 0 or actual_index >= len(self.emails):
            print('Invalid email number.\n')
            return
        self.emails[actual_index].display_full_email()

    def delete_email(self, index):
        if not self.emails:
            print('Inbox is empty.\n')
            return
        actual_index = index - 1
        if actual_index < 0 or actual_index >= len(self.emails):
            print('Invalid email number.\n')
            return
        del self.emails[actual_index]
