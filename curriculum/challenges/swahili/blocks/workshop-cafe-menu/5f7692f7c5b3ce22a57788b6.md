assert.lengthOf(document.querySelectorAll('p.flavor'), 5);

const flavor = document.querySelectorAll('p.flavor');

assert.match(flavor?.[2]?.innerText.trim(), /Pumpkin Spice/i);
assert.match(flavor?.[3]?.innerText.trim(), /Hazelnut/i);
