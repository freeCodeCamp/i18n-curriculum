`if` 文の後に、右のサブツリーを検索するために `self._search` を `node.right` と `key` で呼び出した結果を返してください。

class BinarySearchTree:

        self.root = None

    def _insert(self, node, key):
        if node is None:
            return TreeNode(key)
        if key < node.key:
            node.left = self._insert(node.left, key)
        elif key > node.key:
            node.right = self._insert(node.right, key)
