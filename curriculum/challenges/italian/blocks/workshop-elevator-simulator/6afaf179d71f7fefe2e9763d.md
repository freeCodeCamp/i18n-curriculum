```py
def run_elevator(requests, start_floor):
    current_floor = start_floor
    floors_traveled = 0

    while requests:
        target = requests.pop(0)

        while current_floor != target:
            if current_floor < target:
                current_floor += 1
            else:
                current_floor -= 1
            floors_traveled += 1

        print(f'Arrived at floor {current_floor}')
```
